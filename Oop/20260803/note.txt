special operators
-----------------
1.+=
    int a = 5;
    a += 5; // a = a + 5
2. -=
    int a = 5;
    a -= 5; // a = a - 5
3. *=
    int a = 5;
    a *= 5; // a = a * 5
4. /=
    int a = 5;
    a /= 5; // a = a / 5

Very Special operators
-----------------------
we dont use thrse on normal business applications, but in like IOT projects etc....

1. ~ ( Compliment operator)
    int a = 5;
    a = ~a; // a = -6
2. & (Bitwise AND operator)
    int a = 5; // 0101
    int b = 3; // 0011
    int c = a & b; // c = 1 (0001)
3. | (Bitwise OR operator)
    int a = 5; // 0101
    int b = 3; // 0011
    int c = a | b; // c = 7 (0111)
4. ^ (Bitwise XOR operator)
    int a = 5; // 0101
    int b = 3; // 0011
    int c = a ^ b; // c = 6 (0110)
5. << (Left shift operator)
    int a = 5; // 0101
    int b = a << 1; // b = 10 (1010)
6 >> (Right shift operator)
    int a = 5; // 0101
    int b = a >> 1; // b = 2 (0010)
7 >>> (Unsigned right shift operator)
    int a = -5; // 11111111 11111111 11111111 11111011
    int b = a >>> 1; // b = 2147483645 (01111111 11111111 11111111 1111101)
